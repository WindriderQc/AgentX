     const QDRANT_URL = process.env.QDRANT_URL || 'http://localhost:6333';
     const VECTOR_STORE_TYPE = process.env.VECTOR_STORE_TYPE;

     if (VECTOR_STORE_TYPE !== 'qdrant') {

         message: 'Qdrant not configured as vector store',
         vectorStoreType: VECTOR_STORE_TYPE || 'memory',
         healthy: false
       });

       if (response.ok) {
         res.json({
           status: 'connected',
           message: 'Vector database operational',
           url: QDRANT_URL,
           vectorStoreType: VECTOR_STORE_TYPE,
           healthy: true,
           lastCheck: new Date().toISOString()
         });
       }
     } catch (err) {
       res.json({
         status: 'error',
         message: err.message,
         url: QDRANT_URL,
         healthy: false
       });
     }
   });

   try {
     const qdrantHealthResponse = await fetch(`${AGENTX_BASE}/api/health/qdrant`);
     const qdrantHealth = await qdrantHealthResponse.json();
     const qdrantHealthy = qdrantHealth.healthy === true;

     // Add Qdrant to health checks
     healthData.checks.push({
       component: 'qdrant',
       status: qdrantHealthy ? 'pass' : 'fail',
       message: qdrantHealth.message || 'Vector database status unknown',
       endpoint: qdrantHealth.url || 'http://localhost:6333',
       details: qdrantHealth.vectorStoreType ? `Vector Store: ${qdrantHealth.vectorStoreType}` : null
     });

     // Update health percentage if Qdrant is down
     if (!qdrantHealthy && healthData.summary) {
       const totalChecks = healthData.checks.length;
       const passedChecks = healthData.checks.filter(c => c.status === 'pass').length;
       healthData.summary.health_percentage = Math.round((passedChecks / totalChecks) * 100);
     }
   } catch (qdrantErr) {
     healthData.checks.push({
       component: 'qdrant',
       status: 'fail',
       message: 'Could not retrieve Qdrant status from AgentX'
     });
   }
